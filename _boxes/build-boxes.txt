INST=1

TMP=${TMP:-/tmp/gfs}
OUTPUT=${OUTPUT:-/tmp}

ROOT=$(pwd)

# Check for duplicate sources (default: OFF)
CHECKDUPLICATE=0


src=(
#osinfo-db-tools -Already built
#osinfo-db  -Already built
#libosinfo -Already built
yajl
numactl
libvirt
libvirt-glib
libvirt-python
gtk-vnc
spice-protocol
spice
usbredir
libcacard
phodav
spice-gtk
device-tree-compiler
libnfs
snappy
vde2
virglrenderer
qemu
virt-manager
libovf-glib
gnome-boxes
)

for dir in ${src[@]}; do

	# get package name
	package=$(echo $dir)

	# Change to package directory
  	cd $ROOT/$dir || exit 1

	# Get the version
  	version=$(cat ${package}.SlackBuild | grep "VERSION:" | head -n1 | cut -d "-" -f2 | rev | cut -c 2- | rev)

	# Get the build
	build=$(cat ${package}.SlackBuild | grep "BUILD:" | cut -d "-" -f2 | rev | cut -c 2- | rev)

  	if [ $CHECKDUPLICATE -eq 1 ]; then
    	# Check for duplicate sources
    	sourcefile="$(ls -l $ROOT/$dir/${package}-*.tar.?z* 2>/dev/null | wc -l)"
    	if [ $sourcefile -gt 1 ]; then
          echo "You have following duplicate sources:"
          ls $ROOT/$dir/${package}-*.tar.?z* | cut -d " " -f1
          echo "Please delete sources other than ${package}-$version to avoid problems"
          exit 1
    	  fi
  	fi


	# The real build starts here
  	TMP=$TMP OUTPUT=$OUTPUT sh ${package}.SlackBuild || exit 1
  	if [ "$INST" = "1" ]; then
    	PACKAGE=$(ls $OUTPUT/${package}-${version}-*-${build}*gfs.txz 2>/dev/null)
    	if [ -f "$PACKAGE" ]; then
      	upgradepkg --install-new --reinstall "$PACKAGE"
    	else
      	echo "Error:  package to upgrade "$PACKAGE" not found in $OUTPUT"
      	exit 1
    	  fi
  	fi

  	# back to original directory
  	cd $ROOT
done
